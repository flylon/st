                WITH IndicatorParsed AS (

                        date,

                        -- 均線解析
                        CAST(json_extract(m_ma, '$.ma5') AS REAL) AS ma5,
                        CAST(json_extract(m_ma, '$.ma10') AS REAL) AS ma10,
                        CAST(json_extract(m_ma, '$.ma20') AS REAL) AS ma20,
                        LAG(CAST(json_extract(m_ma, '$.ma20') AS REAL)) OVER (PARTITION BY code ORDER BY date) AS prev_ma20,

                        CAST(json_extract(s_macd, '$.osc') AS REAL) AS osc,
                        LAG(CAST(json_extract(s_macd, '$.osc') AS REAL)) OVER (PARTITION BY code ORDER BY date) AS prev_osc,

                        -- KD 解析
                        CAST(json_extract(s_kd, '$.k') AS REAL) AS k_val,
                        CAST(json_extract(s_kd, '$.d') AS REAL) AS d_val,
                        LAG(CAST(json_extract(s_kd, '$.k') AS REAL)) OVER (PARTITION BY code ORDER BY date) AS prev_k,
                        LAG(CAST(json_extract(s_kd, '$.d') AS REAL)) OVER (PARTITION BY code ORDER BY date) AS prev_d

                    FROM stock_daily
                ),
                PreCalculated AS (
                    SELECT
                        *,
                        -- 在此層明確算出 3 條均線的最大值與最小值，避免 WHERE 誤判為聚合
                        CASE
                            WHEN ma5 >= ma10 AND ma5 >= ma20 THEN ma5
                            WHEN ma10 >= ma5 AND ma10 >= ma20 THEN ma10
                            ELSE ma20
                        END AS ma_max,
                        CASE
                            WHEN ma5 <= ma10 AND ma5 <= ma20 THEN ma5
                            WHEN ma10 <= ma5 AND ma10 <= ma20 THEN ma10
                            ELSE ma20
                        END AS ma_min
                    FROM IndicatorParsed
                )
                SELECT
                    date,
                    code,
                    name,
                    close_p,
                    ma20,
                    ROUND(vol_cur / 1000, 0) AS vol_cur_k,

                    prev_osc,
                    osc,

* **邏輯：** 這過濾掉了暴漲暴跌的妖股，只留下底部墊高、沒有發生過恐慌性拋售 (跌停) 的強勢股。例如某股，累積 4 次漲停且 0 次跌停。
#### 第二層過濾：K 線型態與均線多頭排列
* **條件：** 股價必須站穩所有短期均線，且 `MA5 > MA20 > MA60` 呈現完美多頭排列發散。
* **邏輯：** 這代表短、中、長期持有的投資人都在獲利狀態，上方沒有套牢的解套賣壓，股價推升的阻力最小。
#### 第三層過濾：動能指標的極端趨勢 (關鍵中的關鍵)
系統使用了 **DMI (趨向指標)** 中的 **ADX (平均趨向指數)** 來判斷趨勢強度。
* **邏輯：** 一般來說，ADX 大於 25 就代表趨勢形成。如系統抓到某股，ADX 大於 25。
當 ADX 極高且 `+DI` 遠大於 `-DI` 時，代表股價正在「主升段噴出」，這種物理慣性在隔天很難瞬間煞車，
這就是準確預測隔天上漲的核心原因。
#### 第四層過濾：單日主力籌碼的「異常放大」
* **條件：** 觀察 OBV (能量潮) 搭配外資/投信的買超數據。
* **邏輯：** 某股的收盤價與前一天持平，但系統發現當天外資「異常」大買了 xx萬張，且 OBV 呈現巨量正值。這暗示主力在吃貨洗盤，因此系統大膽預測 1~3個交易日 會噴出，結果 1~3個交易日 果然跳空放量大漲。
