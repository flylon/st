                            ELSE ma20
                        END AS ma_min

                    date,
                    code,
                    name,
                    close_p,

                    osc,
                    k_val,
                    d_val
                FROM PreCalculated

                    -- date 欄位可能是 INTEGER，而 debug SQL 將日期代入後是文字常值；

                    CAST(date AS TEXT) = CAST('20261019' AS TEXT)
                    AND code GLOB '[0-9][0-9][0-9][0-9]'
                    AND vol_cur >= 500000

                    AND ((ma_max - ma_min) / ma_min) <= 0.035
                    AND ma20 >= prev_ma20
                    AND close_p >= ma20
                    AND ((close_p - ma20) / ma20) <= 0.04

                    -- 量能與動能指標
                    AND vol_cur >= (vol_ma5 * 1.2)
                    -- AND prev_osc <= 0

                    AND k_val >= 30
                    AND k_val <= 45
