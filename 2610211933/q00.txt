                        CAST(json_extract(m_ma, '$.ma20') AS REAL) AS ma20,
                        LAG(CAST(json_extract(m_ma, '$.ma20') AS REAL)) OVER (PARTITION BY code ORDER BY date) AS prev_ma20,

                        -- 成交量解析
                        CAST(json_extract(s_vol, '$.current') AS REAL) AS vol_cur,
                        CAST(json_extract(s_vol, '$.ma5') AS REAL) AS vol_ma5,

                        -- MACD 柱狀體解析
                        CAST(json_extract(s_macd, '$.osc') AS REAL) AS osc,
                        LAG(CAST(json_extract(s_macd, '$.osc') AS REAL)) OVER (PARTITION BY code ORDER BY date) AS prev_osc,

                        -- KD 解析
                        CAST(json_extract(s_kd, '$.k') AS REAL) AS k_val,
                        CAST(json_extract(s_kd, '$.d') AS REAL) AS d_val,
                        LAG(CAST(json_extract(s_kd, '$.k') AS REAL)) OVER (PARTITION BY code ORDER BY date) AS prev_k,
                        LAG(CAST(json_extract(s_kd, '$.d') AS REAL)) OVER (PARTITION BY code ORDER BY date) AS prev_d

                    FROM otc_stock_daily
                ),
                PreCalculated AS (
                    SELECT
                        *,
                        -- 在此層明確算出 3 條均線的最大值與最小值，避免 WHERE 誤判為聚合
                        CASE
                            WHEN ma5 >= ma10 AND ma5 >= ma20 THEN ma5
                            WHEN ma10 >= ma5 AND ma10 >= ma20 THEN ma10
                            ELSE ma20
                        END AS ma_max,
                        CASE
                            WHEN ma5 <= ma10 AND ma5 <= ma20 THEN ma5
                            WHEN ma10 <= ma5 AND ma10 <= ma20 THEN ma10
                            ELSE ma20
                        END AS ma_min
                    FROM IndicatorParsed
                )
                SELECT
                    date,
                    code,
                    name,
                    close_p,
                    ma20,
                    ROUND(vol_cur / 1000, 0) AS vol_cur_k,
                    ROUND(vol_ma5 / 1000, 0) AS vol_ma5_k,
                    prev_osc,
                    osc,
                    k_val,
                    d_val
                FROM PreCalculated
                WHERE
                    -- date 欄位可能是 INTEGER，而 debug SQL 將日期代入後是文字常值；
                    -- 統一轉為文字，確保參數化執行與直接執行 debug SQL 結果一致。
                    CAST(date AS TEXT) = CAST('20261021' AS TEXT)
                    AND code GLOB '[0-9][0-9][0-9][0-9]'
                    AND vol_cur >= 500000
                    -- 均線糾結度判斷
                    AND ((ma_max - ma_min) / ma_min) <= 0.035
                    AND ma20 >= prev_ma20
                    AND close_p >= ma20
                    AND ((close_p - ma20) / ma20) <= 0.04

                    -- 量能與動能指標
                    AND vol_cur >= (vol_ma5 * 1.2)
                    -- AND prev_osc <= 0
                    -- AND osc >= 0

                    -- KD 條件
                    AND k_val > d_val
                    AND k_val >= 30
                    AND k_val <= 45
                ORDER BY code ASC;
