
-- 01.建議觀查賣出的數量統計 --
[SQL]
WITH x_values(x) AS (
    VALUES (80), (85), (90),(95),(97),(99),(100)
)
SELECT
    CAST(x AS TEXT) AS n,
    (
        SELECT COUNT(*)
        FROM "stock_daily"
        WHERE 1=1
          AND date = strftime('%Y%m%d', 'now', '+8 hours')
          AND s_kdj IS NOT NULL
          AND json_valid(s_kdj)
          AND json_extract(s_kdj, '$.j') >= x_values.x
          AND s_rsi IS NOT NULL
          AND json_valid(s_rsi)
          AND json_extract(s_rsi, '$.rsi6') >= x_values.x
    ) AS cnts
FROM x_values;
+-----+------+
| n   | cnts |
+-----+------+
| 80  | 0    |
| 85  | 0    |
| 90  | 0    |
| 95  | 0    |
| 97  | 0    |
| 99  | 0    |
| 100 | 0    |

-- 02.建議觀查的買進數量統計 --
[SQL]
WITH x_values(x) AS (
    VALUES (1),(2),(3),(4),(5),(7),(9), (10), (15), (20), (25), (30),(35)
)
SELECT
    CAST(x AS TEXT) AS n,
    (
        SELECT COUNT(*)
        FROM "stock_daily"
        WHERE 1=1
          AND date = strftime('%Y%m%d', 'now', '+8 hours')
          AND s_kdj IS NOT NULL
          AND json_valid(s_kdj)
          AND json_extract(s_kdj, '$.j') <= x_values.x
          AND s_rsi IS NOT NULL
          AND json_valid(s_rsi)
          AND json_extract(s_rsi, '$.rsi6') <= x_values.x

FROM x_values;
